### `FormationSet.FromString` 的四条规则（逐行原文）

```csharp
if (formationString == "") { IsSelfTarget = true; IsSelfFormation = true; return; }   // ① 空串
while (formationString[0] == '@' || formationString[0] == '~' || formationString[0] == '?') {
    if (formationString[0] == '@') IsSelfFormation = true;    // ② 己方阵营
    if (formationString[0] == '~') IsMultitarget   = true;    // ③ 多目标
    if (formationString[0] == '?') IsRandomTarget  = true;    // ④ 随机
    formationString = formationString.Substring(1);
}
for (int i = 0; i < formationString.Length; i++)
    Ranks.Add(int.Parse(formationString[i].ToString()));       // 🔴 ⑤ 逐字符 = 一个 rank
Ranks.Sort();
```
